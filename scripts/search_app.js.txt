(function () {
  'use strict';
  var S = window.TopdaScore;
  var form = document.getElementById('finderForm');
  var resultsEl = document.getElementById('finderResults');
  var countEl = document.getElementById('finderCount');
  var relaxedEl = document.getElementById('finderRelaxed');
  var sidoSel = document.getElementById('f-sido');
  var guSel = document.getElementById('f-gu');
  var familySel = document.getElementById('f-family');
  var areaSel = document.getElementById('f-area');
  var searchInp = document.getElementById('f-search');
  var sortSel = document.getElementById('f-sort');
  var DATA = [];
  var DEALS = [];

  // 국제 통화 표기: zh 는 亿(=억), 그 외는 십억(₩ B/tỷ/พันล้าน).
  function fmt(manwon) {
    if (manwon == null) return '—';
    var eok = manwon / 10000;
    if (T.curMode === 'eok') {
      var v = Number.isInteger(eok) ? eok : Math.round(eok * 10) / 10;
      return T.curPrefix + v + T.curUnit;
    }
    var bn = Math.round((eok / 10) * 100) / 100;
    return T.curPrefix + bn + T.curUnit;
  }
  function esc(s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
  function num(v) { v = parseFloat(v); return isNaN(v) ? null : v; }
  function hubName(k) { return (typeof HUB_DISPLAY !== 'undefined' && HUB_DISPLAY[k]) || k; }

  // 한글 지역명 → 로마자(개정 로마자 표기 근사). 값(필터)은 한글 유지, 표시만 로마자.
  var RM_CHO = ['g','kk','n','d','tt','r','m','b','pp','s','ss','','j','jj','ch','k','t','p','h'];
  var RM_JUNG = ['a','ae','ya','yae','eo','e','yeo','ye','o','wa','wae','oe','yo','u','wo','we','wi','yu','eu','ui','i'];
  var RM_JONG = ['','k','k','k','n','n','n','t','l','k','m','l','l','l','p','l','m','p','p','t','t','ng','t','t','k','t','p','t'];
  var RM_SUFFIX = { '구': 'gu', '군': 'gun', '시': 'si', '동': 'dong', '읍': 'eup', '면': 'myeon', '도': 'do' };
  function romanizeSyl(ch) {
    var code = ch.charCodeAt(0) - 0xAC00;
    if (code < 0 || code > 11171) return ch;
    return RM_CHO[Math.floor(code / 588)] + RM_JUNG[Math.floor((code % 588) / 28)] + RM_JONG[code % 28];
  }
  function romanizeWord(w) {
    if (!w) return w;
    // 행정 접미사(구/군/시/동…)는 어간이 2음절 이상일 때만 분리한다.
    // (대구=Daegu 처럼 접미사가 아닌 1음절 어간 지명이 'Dae-gu'로 잘못 쪼개지는 것 방지)
    var last = w.slice(-1), stem = w, suf = '';
    if (RM_SUFFIX[last] && w.length >= 3) { stem = w.slice(0, -1); suf = '-' + RM_SUFFIX[last]; }
    var r = '';
    for (var i = 0; i < stem.length; i++) r += romanizeSyl(stem[i]);
    if (r) r = r.charAt(0).toUpperCase() + r.slice(1);
    return r + suf;
  }
  function romanize(s) { return s ? String(s).split(' ').map(romanizeWord).join(' ') : s; }
  // 평형 라벨의 한글 '평'을 현지 단위로 치환 (데이터 라벨 예: "85㎡(26평)")
  function unitLabel(s) { return s ? String(s).replace(/\((\d+)\s*평\)/g, function (_, n) { return '(' + n + ' ' + T.py + ')'; }) : s; }

  function readFilters() {
    var f = {};
    var budget = num(form.budget.value);
    if (budget) f.budget_max = Math.round(budget * (T.budgetScale || 10000));
    if (guSel.value && guSel.value !== 'all') f.region = guSel.value;
    else if (sidoSel.value && sidoSel.value !== 'all') f.region = sidoSel.value;
    if (areaSel.value && areaSel.value !== 'all') f.area_type = areaSel.value;
    var age = num(form.max_age_years.value); if (age) f.max_age_years = age;
    var sub = num(form.subway_distance_max.value); if (sub) f.subway_distance_max = sub;
    var elem = num(form.elementary_distance_max.value); if (elem) f.elementary_distance_max = elem;
    var house = num(form.min_households.value); if (house) f.min_households = house;
    if (form.commute_hub.value) {
      f.commute_hub = form.commute_hub.value;
      f.commute_max = num(form.commute_max.value) || 60;
    }
    var park = num(form.parking_min.value); if (park) f.parking_min = park;
    var ratio = num(form.jeonse_ratio_max.value); if (ratio) f.jeonse_ratio_max = ratio;
    var q = (searchInp.value || '').trim();
    if (q) f._search = q.toLowerCase();
    return f;
  }

  function displayName(a) { return (a && (a.display_name || a.name)) || ''; }
  function matchesSearch(a, q) {
    if (!q) return true;
    var hay = ((a.name || '') + ' ' + displayName(a) + ' ' + (a.region || '') + ' ' +
      (a.region_key || '') + ' ' + (a.builder || '')).toLowerCase();
    return hay.indexOf(q) >= 0;
  }

  function comparator(mode) {
    if (mode === 'price_asc')  return function (a, b) { return (a.price || Infinity) - (b.price || Infinity); };
    if (mode === 'price_desc') return function (a, b) { return (b.price || 0) - (a.price || 0); };
    if (mode === 'recent') {
      return function (a, b) {
        var ad = (DEALS_BY_APT[a.apt.name] || {}).latest || '';
        var bd = (DEALS_BY_APT[b.apt.name] || {}).latest || '';
        return bd < ad ? -1 : bd > ad ? 1 : b.total - a.total;
      };
    }
    if (mode === 'momentum') {
      return function (a, b) {
        var am = S.scoreMomentum(a.apt.price_history) || 0;
        var bm = S.scoreMomentum(b.apt.price_history) || 0;
        return bm - am;
      };
    }
    return function (a, b) { return b.total - a.total; };
  }

  var DEALS_BY_APT = {};

  function enrichWithDeals() {
    if (!DEALS.length) return;
    var byName = {};
    DATA.forEach(function (a) { byName[a.name] = a; });
    var newApts = {};
    DEALS.forEach(function (d) {
      var apt = byName[d.apt];
      if (!apt) {
        if (!newApts[d.apt]) {
          var rk = d.region_key || '';
          var rkParts = rk.split(' ');
          var sido = d.sido || rkParts[0] || '';
          var sigungu = rkParts.slice(1).join(' ');
          newApts[d.apt] = {
            name: d.apt, region: d.region || rk, region_key: rk,
            sido: sido, sigungu: sigungu, built_year: d.build_year || null,
            units: [], price_history: [], _from_deals: true,
          };
        }
        apt = newApts[d.apt];
      }
      var key = d.apt;
      if (!DEALS_BY_APT[key]) DEALS_BY_APT[key] = { latest: '', prices: [] };
      var agg = DEALS_BY_APT[key];
      if (d.date > agg.latest) agg.latest = d.date;
      agg.prices.push({ area_m2: d.area_m2, price: d.price, date: d.date });
      if (!apt.units || !apt.units.length) {
        if (!apt._unitMap) apt._unitMap = {};
        var band = Math.round(d.area_m2);
        var slot = apt._unitMap[band];
        if (!slot || slot.date < d.date) {
          apt._unitMap[band] = { area_m2: band, recent_price: d.price, date: d.date, label: band + '㎡' };
        }
      }
    });
    Object.keys(newApts).forEach(function (n) {
      var a = newApts[n];
      if (a._unitMap) {
        a.units = Object.keys(a._unitMap).map(function (k) { return a._unitMap[k]; })
          .sort(function (x, y) { return x.area_m2 - y.area_m2; })
          .map(function (u) { return { label: u.label, area_m2: u.area_m2, recent_price: u.recent_price }; });
        delete a._unitMap;
      }
      var allPrices = (DEALS_BY_APT[n] && DEALS_BY_APT[n].prices || []).filter(function (p) { return p.area_m2 > 0; });
      var areas = allPrices.map(function (p) { return p.area_m2; }).sort(function (x, y) { return x - y; });
      var repArea = areas.length ? areas[Math.floor(areas.length / 2)] : 0;
      var months = {};
      allPrices.forEach(function (p) {
        var mo = p.date.slice(0, 7);
        if (!months[mo]) months[mo] = [];
        months[mo].push(p.price / p.area_m2);
      });
      a.price_history = Object.keys(months).sort().slice(-12).map(function (mo) {
        var arr = months[mo]; return Math.round(arr.reduce(function (s, x) { return s + x; }, 0) / arr.length * repArea);
      });
      var cutoff = new Date(Date.now() - 180 * 86400000).toISOString().slice(0, 10);
      var recent = (DEALS_BY_APT[n] && DEALS_BY_APT[n].prices || []).filter(function (p) { return p.date >= cutoff; });
      a.recent_tx_count = recent.length;
      DATA.push(a);
    });
    DATA.forEach(function (a) {
      if (a.recent_tx_count == null && DEALS_BY_APT[a.name]) {
        var cut = new Date(Date.now() - 180 * 86400000).toISOString().slice(0, 10);
        a.recent_tx_count = DEALS_BY_APT[a.name].prices.filter(function (p) { return p.date >= cut; }).length;
      }
    });
  }
  var curHub = '';

  function barRow(label, val) {
    if (val == null) return '';
    var w = Math.round(val);
    return '<div class="apt-bar"><span>' + label + '</span><span class="track"><span class="fill" style="width:' + w + '%"></span></span><span>' + w + '</span></div>';
  }

  function latestDealOf(aptName) { var x = DEALS_BY_APT[aptName]; return x ? x.latest : ''; }
  function daysSince(s) { if (!s) return null; var d = new Date(s); if (isNaN(d)) return null; return Math.floor((Date.now() - d.getTime()) / 86400000); }
  function ago(days) {
    if (days == null) return '';
    if (days < 1) return T.ago.today;
    if (days < 30) return days + T.ago.d;
    if (days < 365) return Math.floor(days / 30) + T.ago.mo;
    return Math.floor(days / 365) + T.ago.y;
  }

  // 구조화 추천 이유(reasonData) → 현지어 칩
  function reasonChip(d) {
    if (!d || !d.key) return '';
    var tpl = T.reasons[d.key];
    if (!tpl) return '';
    if (d.key === 'subway') return tpl.replace('{station}', esc(d.station)).replace('{min}', d.min);
    if (d.key === 'school') return tpl.replace('{name}', esc(d.name)).replace('{dist}', d.dist).replace('{zone}', d.inZone ? T.assigned : '');
    if (d.key === 'scale') return tpl.replace('{n}', Number(d.households).toLocaleString());
    if (d.key === 'age') return tpl.replace('{year}', d.year);
    if (d.key === 'commute') return tpl.replace('{hub}', esc(hubName(d.hub))).replace('{min}', d.min);
    if (d.key === 'parking') return tpl.replace('{per}', Number(d.per).toFixed(2));
    return tpl; // momentum, stability
  }

  function card(r, rank) {
    var a = r.apt;
    var rd = r.reasonData || [];
    var reasons = rd.map(function (d) { var x = reasonChip(d); return x ? '<span class="apt-chip">' + x + '</span>' : ''; }).join('');
    if (!reasons && r.bestKey) reasons = '<span class="apt-chip">' + ((T.bestLabels[r.bestKey] || r.bestKey) + T.bestSuffix) + '</span>';
    var relaxed = (r.relaxedKeys && r.relaxedKeys.length)
      ? '<div class="apt-relaxed">' + T.relaxedCardPre + esc(r.relaxedKeys.map(function (k) { return T.filterLabels[k] || k; }).join(', ')) + T.relaxedCardSuf + '</div>'
      : '';
    var bars = (r.sub.fit != null ? barRow(T.bars.fit, r.sub.fit) : '')
      + barRow(T.bars.appreciation, r.sub.appreciation) + barRow(T.bars.subway, r.sub.subway) + barRow(T.bars.school, r.sub.school)
      + barRow(T.bars.stability, r.sub.stability) + barRow(T.bars.liquidity, r.sub.liquidity)
      + barRow(T.bars.scale, r.sub.scale) + barRow(T.bars.age, r.sub.age);
    var sub = a.subway && a.subway.distance_m != null
      ? a.subway.distance_m + 'm · ' + esc(a.subway.station || '') + ' (' + T.walk + ' ' + S.walkMin(a.subway.distance_m) + T.min + (a.subway.line ? ', ' + esc(a.subway.line) : '') + ')'
      : T.noInfo;
    var elem = a.elementary && a.elementary.distance_m != null
      ? a.elementary.distance_m + 'm · ' + esc(a.elementary.name || '')
      : T.noInfo;
    var house = a.households ? a.households.toLocaleString() + T.units : T.unitsUnknown;
    if (a.builder && a.builder !== '기타') house += (a.households ? ' · ' : '') + esc(a.builder);
    var latest = latestDealOf(a.name);
    var repLabel = esc(unitLabel(r.rep ? r.rep.label : ''));
    var dealNote = latest ? repLabel + ' · ' + latest + ' (' + ago(daysSince(latest)) + ')' : repLabel + ' ' + T.recentDeal;
    return '' +
      '<article class="apt-card" data-apt="' + esc(a.name) + '" data-region="' + esc(a.region_key || '') + '">' +
        '<span class="rank">' + T.rankPre + rank + T.rankSuf + '</span>' +
        '<div class="apt-head">' +
          '<div><div class="apt-name">' + esc(displayName(a)) + '</div><div class="apt-region">' + esc(romanize(a.region)) + '</div></div>' +
          '<div class="apt-score"><div class="num">' + r.total + '</div><div class="lbl">' + T.totalScore + '</div></div>' +
        '</div>' +
        '<div class="apt-price">' + fmt(r.price) + ' <small>' + dealNote + '</small></div>' +
        '<div class="apt-units">' + (a.units || []).map(function (u) {
          return '<span class="apt-unit' + (r.rep && u.area_m2 === r.rep.area_m2 ? ' is-rep' : '') + '">' + esc(unitLabel(u.label)) + ' ' + fmt(u.recent_price) + '</span>';
        }).join('') + '</div>' +
        relaxed +
        '<div class="apt-spec">' +
          '<span class="k">' + T.specSubway + '</span><span class="v">' + sub + '</span>' +
          '<span class="k">' + T.specElem + '</span><span class="v">' + elem + '</span>' +
          '<span class="k">' + T.specBuilt + '</span><span class="v">' + (a.built_year ? a.built_year + T.builtYr + ' (' + r.ageYears + T.yrOld + ')' : T.unknown) + '</span>' +
          '<span class="k">' + T.specHouseholds + '</span><span class="v">' + house + '</span>' +
          (curHub && a.commute && a.commute[curHub] != null ? '<span class="k">' + T.toPrefix + esc(hubName(curHub)) + T.toSuffix + '</span><span class="v">' + T.about + a.commute[curHub] + T.min + '</span>' : '') +
          (a.jeonse_ratio != null ? '<span class="k">' + T.specRatio + '</span><span class="v">' + a.jeonse_ratio + '%' + (a._jr_region ? ' <small>' + T.areaAvg + '</small>' : '') + '</span>' : '') +
          (a.parking_per_household != null ? '<span class="k">' + T.specParking + '</span><span class="v">' + a.parking_per_household.toFixed(2) + T.perUnit + '</span>' : '') +
        '</div>' +
        '<div class="apt-reasons">' + reasons + '</div>' +
        '<div class="apt-bars">' + bars + '</div>' +
      '</article>';
  }

  // 결과는 한 번에 다 그리지 않고 PAGE개씩 이어붙인다. 조건이 느슨하면 수백 곳이
  // 잡히는데, 카드 하나가 DOM 수십 노드라 한꺼번에 그리면 모바일에서 눈에 띄게 버벅인다.
  // 잘라내는 게 아니라 나눠 그리는 것이라 전체 결과에는 그대로 접근할 수 있다.
  var PAGE = 50;
  var curList = [], shown = 0;

  function paintMore() {
    var next = curList.slice(shown, shown + PAGE);
    if (!next.length) return;
    var oldBtn = document.getElementById('finderMore');
    if (oldBtn) oldBtn.parentNode.removeChild(oldBtn);
    // 순위 번호는 이어서 매긴다(51번째가 다시 1이 되면 안 된다).
    resultsEl.insertAdjacentHTML('beforeend', next.map(function (r, i) {
      return card(r, shown + i + 1);
    }).join(''));
    shown += next.length;
    var left = curList.length - shown;
    if (left > 0) {
      resultsEl.insertAdjacentHTML('beforeend',
        '<button type="button" class="btn btn-secondary finder-more" id="finderMore">' +
        T.more + '<span class="finder-more-left">' + T.moreLeft.replace('{n}', left) + '</span></button>');
    }
  }

  function showList(list) {
    curList = list;
    shown = 0;
    resultsEl.innerHTML = '';
    paintMore();
  }

  // 결과 없음·안내 문구를 넣을 때도 상태를 비워야 이전 목록의 '더보기'가 되살아나지 않는다.
  function showMessage(html) {
    curList = [];
    shown = 0;
    resultsEl.innerHTML = html;
  }

  // 같은 조건이면 다시 그리지 않는다. 텍스트 입력은 input 이벤트로 이미 반영되는데,
  // 포커스가 빠질 때 change 이벤트가 한 번 더 오기 때문에(예산 칸 입력 후 '더보기' 클릭)
  // 그대로 두면 펼쳐둔 목록이 첫 50곳으로 되감긴다. 불필요한 재계산도 함께 줄인다.
  var lastKey = null;

  function render() {
    if (!DATA.length) return;
    var f = readFilters();
    var q = f._search; delete f._search;
    var mode = sortSel.value || 'score';
    var key = DATA.length + '|' + mode + '|' + (q || '') + '|' + JSON.stringify(f);
    if (key === lastKey) return;
    lastKey = key;
    curHub = f.commute_hub || '';
    var hasFilter = Object.keys(f).length > 0 || !!q;
    if (!hasFilter) {
      relaxedEl.innerHTML = '';
      countEl.textContent = '';
      showMessage('<div class="finder-empty">' + T.initialEmpty + '</div>');
      return;
    }
    var pool = q ? DATA.filter(function (a) { return matchesSearch(a, q); }) : DATA;
    var list = S.search(pool, f);
    list.sort(comparator(mode));
    relaxedEl.innerHTML = '';
    if (list.length) {
      var sortLabel = T.sortLabelsShort[mode] || T.sortLabelsShort.score;
      countEl.innerHTML = T.countMatch.replace('{n}', list.length).replace('{sort}', sortLabel);
      showList(list);
    } else {
      var near = S.nearMatches(pool, f, null, 2);
      if (near.length) {
        countEl.innerHTML = T.countRelaxed;
        relaxedEl.innerHTML = '<div class="finder-relaxed-banner">' + T.relaxedBanner + '</div>';
        showList(near);
      } else {
        countEl.textContent = T.noMatch;
        showMessage('<div class="finder-empty">' + T.emptyRelax + '</div>');
      }
    }
  }

  // ── 단지 상세 팝업 ──
  var aptModal = document.getElementById('aptModal');
  var finderMap = null;   // TopdaKMap 어댑터(카카오 우선, 실패 시 Leaflet 폴백)

  function findApt(name, regionKey) {
    for (var i = 0; i < DATA.length; i++) {
      var a = DATA[i];
      if (a.name === name && (!regionKey || (a.region_key || '') === regionKey)) return a;
    }
    return null;
  }

  // 상세 팝업 지도 — TopdaKMap(카카오 우선, 도메인 미등록/로컬은 Leaflet 폴백).
  // 어댑터를 한 번만 만들고 이후엔 재중심(setCenter)만 해서 모달 재오픈에 재사용한다.
  function showFinderMap(a) {
    var mapEl = document.getElementById('aptModalMap');
    if (a.lat == null || a.lng == null) { mapEl.style.display = 'none'; return; }
    mapEl.style.display = '';
    function place(ad) {
      if (!ad) return;
      finderMap = ad;
      ad.clear();
      if (ad.relayout) ad.relayout();
      ad.setCenter(a.lat, a.lng);
      ad.addCircle(a.lat, a.lng, { fillColor: '#2563eb', radius: 10, title: displayName(a) });
      // 모달이 방금 표시돼 컨테이너 크기가 잡히기 전이면 한 번 더 재배치·재중심.
      setTimeout(function () { if (ad.relayout) ad.relayout(); ad.setCenter(a.lat, a.lng); }, 90);
    }
    if (finderMap) { place(finderMap); return; }
    if (typeof window.TopdaKMap !== 'undefined') {
      window.TopdaKMap.init('aptModalMap', { center: [a.lat, a.lng], level: 3, zoom: 16, maxZoom: 16 }, place);
    }
  }

  function openAptModal(name, regionKey) {
    var a = findApt(name, regionKey);
    if (!a) return;
    document.getElementById('aptModalTitle').textContent = displayName(a);
    document.getElementById('aptModalSub').textContent = romanize(a.region || '');

    var agg = DEALS_BY_APT[a.name] || { prices: [] };
    var deals = agg.prices.slice().sort(function (x, y) { return x.date < y.date ? 1 : -1; });
    var meta = [];
    if (deals.length) {
      var last = deals[0];
      var vals = deals.map(function (d) { return d.price; });
      var avg = Math.round(vals.reduce(function (s, v) { return s + v; }, 0) / vals.length);
      meta.push({ lbl: T.mLatest, val: fmt(last.price) + ' · ' + last.date });
      meta.push({ lbl: T.mDeals, val: deals.length.toLocaleString() + T.dealsCnt });
      meta.push({ lbl: T.mAvg, val: fmt(avg) });
    }
    if (a.built_year) meta.push({ lbl: T.mBuilt, val: a.built_year + T.builtYr });
    if (a.households) meta.push({ lbl: T.mHouseholds, val: a.households.toLocaleString() + T.units });

    render();
  });
